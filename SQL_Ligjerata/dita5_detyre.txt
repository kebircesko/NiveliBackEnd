1. Tabela: Lendet
CREATE TABLE Lendet (
    LendaID INT PRIMARY KEY,
    EmriLendes VARCHAR(100) NOT NULL
);

//LendaID është Primary Key (identifikon unikisht çdo lëndë)

Tabela: Studentet
CREATE TABLE Studentet (
    StudentiID INT PRIMARY KEY,
    Emri VARCHAR(100),
    Mbiemri VARCHAR(100),
    LendaID INT,
    FOREIGN KEY (LendaID) REFERENCES Lendet(LendaID)
);

//StudentiID është Primary Key
//LendaID është Foreign Key që lidhet me tabelën Lendet

Bonus (JOIN për me i parë bashkë)
SELECT 
    Studentet.Emri,
    Studentet.Mbiemri,
    Lendet.EmriLendes
FROM Studentet
JOIN Lendet ON Studentet.LendaID = Lendet.LendaID;
